#!/usr/bin/env node /* ============================================================ Portaal & Countdown Beheer Pro — Server (V3) Echte accounts, beveiligd inloggen, volledig logboek, wachtwoord wijzigen en gebruikersbeheer voor de beheerder. Vereist alleen Node.js (18 of nieuwer). Geen npm install nodig. Starten: node server.js → http://localhost:3000 ============================================================ */ 'use strict'; const http = require('http'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); const PORT = process.env.PORT || 3000; const PUBLIC_DIR = path.join(__dirname, 'public'); const DATA_DIR = path.join(__dirname, 'data'); const TOKEN_TTL_MS = 7 * 24 * 60 * 60 * 1000; // 7 dagen const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; fs.mkdirSync(DATA_DIR, { recursive: true }); /* ================= OPSLAG (JSON-bestanden) ================= */ function loadJson(file, fallback) { try { return JSON.parse(fs.readFileSync(path.join(DATA_DIR, file), 'utf8')); } catch (e) { return fallback; } } function saveJson(file, obj) { fs.writeFileSync(path.join(DATA_DIR, file), JSON.stringify(obj, null, 2)); } let users = loadJson('users.json', []); // { id, email, role, createdAt, pass, tv, lastLogin } let logs = loadJson('logs.json', []); // { t, userId, email, action, ip, ua, detail } let content = loadJson('content.json', {}); // userId -> { countdowns: [], todos: [] } let secret = loadJson('secret.key', null); if (!secret || typeof secret !== 'string' || secret.length < 32) { secret = crypto.randomBytes(48).toString('hex'); saveJson('secret.key', secret); } /* ================= WACHTWOORDEN (scrypt + salt) ================= */ function hashPassword(password) { const salt = crypto.randomBytes(16).toString('hex'); const hash = crypto.scryptSync(password, salt, 64).toString('hex'); return salt + ':' + hash; } function verifyPassword(password, stored) { try { const parts = String(stored).split(':'); if (parts.length !== 2) return false; const salt = parts[0]; const orig = Buffer.from(parts[1], 'hex'); const test = crypto.scryptSync(password, salt, orig.length); return orig.length === test.length && crypto.timingSafeEqual(orig, test); } catch (e) { return false; } } /* ================= SESSIE-TOKENS (HMAC-ondertekend) ================= */ function signToken(payload) { const body = Buffer.from(JSON.stringify(payload)).toString('base64url'); const sig = crypto.createHmac('sha256', secret).update(body).digest('base64url'); return body + '.' + sig; } function verifyToken(token) { try { const parts = String(token).split('.'); if (parts.length !== 2) return null; const expected = crypto.createHmac('sha256', secret).update(parts[0]).digest('base64url'); const a = Buffer.from(parts[1]); const b = Buffer.from(expected); if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) return null; const payload = JSON.parse(Buffer.from(parts[0], 'base64url').toString('utf8')); if (!payload || typeof payload.exp !== 'number' || payload.exp < Date.now()) return null; return payload; } catch (e) { return null; } } function parseCookies(req) { const out = {}; const header = req.headers.cookie; if (!header) return out; header.split(';').forEach(function (part) { const i = part.indexOf('='); if (i > -1) out[part.slice(0, i).trim()] = part.slice(i + 1).trim(); }); return out; } /* Token bevat een versienummer (tv). Bij een wachtwoordwijziging of reset wordt de versie opgehoogd, waardoor oude sessies ongeldig worden. */ function currentUser(req) { const token = parseCookies(req)['pcp_session']; if (!token) return null; const payload = verifyToken(token); if (!payload || !payload.uid) return null; for (const u of users) { if (u.id === payload.uid) { if ((u.tv || 0) !== (payload.tv || 0)) return null; // sessie ingetrokken return u; } } return null; } function issueSession(res, user) { setSessionCookie(res, signToken({ uid: user.id, tv: user.tv || 0, exp: Date.now() + TOKEN_TTL_MS })); } function setSessionCookie(res, token) { res.setHeader('Set-Cookie', 'pcp_session=' + token + '; HttpOnly; Path=/; SameSite=Strict; Max-Age=' + Math.floor(TOKEN_TTL_MS / 1000)); } function clearSessionCookie(res) { res.setHeader('Set-Cookie', 'pcp_session=; HttpOnly; Path=/; SameSite=Strict; Max-Age=0'); } /* ================= INLOGPLOEG-LIMIET (rate limiting) ================= */ const loginAttempts = {}; // sleutel -> { count, first } const MAX_ATTEMPTS = 6; const WINDOW_MS = 15 * 60 * 1000; function tooManyAttempts(key) { const a = loginAttempts[key]; if (!a) return false; if (Date.now() - a.first > WINDOW_MS) { delete loginAttempts[key]; return false; } return a.count >= MAX_ATTEMPTS; } function recordFailedAttempt(key) { const a = loginAttempts[key] || { count: 0, first: Date.now() }; a.count += 1; loginAttempts[key] = a; } function clearAttempts(key) { delete loginAttempts[key]; } /* ================= LOGB0EK ================= */ function clientIp(req) { const fwd = req.headers['x-forwarded-for']; if (typeof fwd === 'string' && fwd.length > 0) return fwd.split(',')[0].trim(); return (req.socket && req.socket.remoteAddress) || 'onbekend'; } function logEvent(user, action, req, detail) { logs.push({ t: new Date().toISOString(), userId: user ? user.id : null, email: user ? user.email : null, action: action, ip: clientIp(req), ua: String(req.headers['user-agent'] || '').slice(0, 200), detail: String(detail || '').slice(0, 200) }); if (logs.length > 5000) logs = logs.slice(-5000); saveJson('logs.json', logs); } /* ================= HULPFUNCTIES ================= */ function uid() { return 'id_' + Date.now().toString(36) + '_' + crypto.randomBytes(5).toString('hex'); } function publicUser(u) { return { id: u.id, email: u.email, role: u.role, createdAt: u.createdAt, lastLogin: u.lastLogin || null }; } function send(res, status, obj) { const body = JSON.stringify(obj); res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' }); res.end(body); } function readBody(req) { return new Promise(function (resolve, reject) { let size = 0; const chunks = []; let aborted = false; req.on('data', function (c) { size += c.length; if (size > 102400) { if (!aborted) { aborted = true; reject(new Error('Verzoek te groot')); } req.destroy(); return; } chunks.push(c); }); req.on('end', function () { if (aborted) return; if (chunks.length === 0) return resolve({}); try { resolve(JSON.parse(Buffer.concat(chunks).toString('utf8'))); } catch (e) { resolve(null); } }); req.on('error', reject); }); } function ensureContent(userId) { if (!content[userId]) { content[userId] = { countdowns: [], todos: [] }; saveJson('content.json', content); } return content[userId]; } function findUserById(id) { for (const u of users) { if (u.id === id) return u; } return null; } function countAdmins() { let n = 0; for (const u of users) if (u.role === 'admin') n += 1; return n; } /* ================= VALIDATIE VAN INHOUD ================= */ function sanitizeCountdowns(list) { if (!Array.isArray(list)) return null; const out = []; for (const item of list.slice(0, 200)) { if (!item || typeof item !== 'object') continue; if (typeof item.id !== 'string' || item.id.length > 60) continue; if (typeof item.title !== 'string') continue; const target = new Date(item.target); if (isNaN(target.getTime())) continue; out.push({ id: item.id, title: item.title.slice(0, 60), target: target.toISOString() }); } return out; } function sanitizeTodos(list) { if (!Array.isArray(list)) return null; const out = []; for (const item of list.slice(0, 500)) { if (!item || typeof item !== 'object') continue; if (typeof item.id !== 'string' || item.id.length > 60) continue; if (typeof item.text !== 'string') continue; out.push({ id: item.id, text: item.text.slice(0, 100), done: !!item.done }); } return out; } /* ================= API: ACCOUNTS ================= */ async function apiRegister(req, res) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : ''; const password = typeof body.password === 'string' ? body.password : ''; if (!EMAIL_RE.test(email) || email.length > 254) { return send(res, 400, { error: 'Vul een geldig e-mailadres in.' }); } if (password.length < 8 || password.length > 100) { return send(res, 400, { error: 'Het wachtwoord moet minimaal 8 tekens zijn.' }); } for (const u of users) { if (u.email === email) return send(res, 409, { error: 'Er bestaat al een account met dit e-mailadres.' }); } const isFirst = users.length === 0; const user = { id: uid(), email: email, role: isFirst ? 'admin' : 'user', createdAt: new Date().toISOString(), lastLogin: null, tv: 0, pass: hashPassword(password) }; users.push(user); saveJson('users.json', users); ensureContent(user.id); logEvent(user, 'register', req, isFirst ? 'Eerste account aangemaakt (beheerder)' : 'Account aangemaakt'); issueSession(res, user); send(res, 200, { user: publicUser(user) }); } async function apiLogin(req, res) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : ''; const password = typeof body.password === 'string' ? body.password : ''; const key = email + '|' + clientIp(req); if (tooManyAttempts(key)) { return send(res, 429, { error: 'Te veel mislukte pogingen. Probeer het over 15 minuten opnieuw.' }); } let user = null; for (const u of users) { if (u.email === email) { user = u; break; } } if (!user || !verifyPassword(password, user.pass)) { recordFailedAttempt(key); logEvent(user, 'login_failed', req, 'Mislukte poging voor: ' + (email || 'onbekend')); return send(res, 401, { error: 'E-mailadres of wachtwoord is onjuist.' }); } clearAttempts(key); user.lastLogin = new Date().toISOString(); saveJson('users.json', users); logEvent(user, 'login', req, 'Succesvol ingelogd'); issueSession(res, user); send(res, 200, { user: publicUser(user) }); } function apiLogout(req, res, user) { if (user) logEvent(user, 'logout', req, 'Uitgelogd'); clearSessionCookie(res); send(res, 200, { ok: true }); } /* Wachtwoord wijzigen (eigen account) */ async function apiChangePassword(req, res, user) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const current = typeof body.current === 'string' ? body.current : ''; const next = typeof body.next === 'string' ? body.next : ''; if (!verifyPassword(current, user.pass)) { return send(res, 401, { error: 'Je huidige wachtwoord is onjuist.' }); } if (next.length < 8 || next.length > 100) { return send(res, 400, { error: 'Het nieuwe wachtwoord moet minimaal 8 tekens zijn.' }); } if (next === current) { return send(res, 400, { error: 'Het nieuwe wachtwoord mag niet hetzelfde zijn als het oude.' }); } user.pass = hashPassword(next); user.tv = (user.tv || 0) + 1; // alle bestaande sessies ongeldig maken saveJson('users.json', users); logEvent(user, 'password_changed', req, 'Wachtwoord gewijzigd door gebruiker'); issueSession(res, user); // meteen opnieuw inloggen met nieuwe sessie send(res, 200, { ok: true }); } /* ================= API: GEBRUIKERSBEHEER (alleen beheerder) ================= */ function requireAdmin(res, user) { if (!user || user.role !== 'admin') { send(res, 403, { error: 'Alleen de beheerder mag dit.' }); return false; } return true; } function apiListUsers(req, res, admin) { const list = users.slice().map(publicUser); send(res, 200, { users: list }); } async function apiSetRole(req, res, admin) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const target = findUserById(String(body.userId || '')); const role = String(body.role || ''); if (!target) return send(res, 404, { error: 'Gebruiker niet gevonden.' }); if (role !== 'admin' && role !== 'user') return send(res, 400, { error: 'Ongeldige rol.' }); if (target.id === admin.id) return send(res, 400, { error: 'Je kunt je eigen rol niet wijzigen.' }); if (target.role === 'admin' && role === 'user' && countAdmins() === 1) { return send(res, 400, { error: 'Er moet altijd minimaal één beheerder blijven.' }); } target.role = role; target.tv = (target.tv || 0) + 1; // sessies van deze gebruiker ongeldig maken saveJson('users.json', users); logEvent(admin, 'role_changed', req, target.email + ' is nu ' + (role === 'admin' ? 'beheerder' : 'gebruiker')); send(res, 200, { ok: true, user: publicUser(target) }); } async function apiAdminResetPassword(req, res, admin) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const target = findUserById(String(body.userId || '')); const newPassword = typeof body.newPassword === 'string' ? body.newPassword : ''; if (!target) return send(res, 404, { error: 'Gebruiker niet gevonden.' }); if (newPassword.length < 8 || newPassword.length > 100) { return send(res, 400, { error: 'Het nieuwe wachtwoord moet minimaal 8 tekens zijn.' }); } target.pass = hashPassword(newPassword); target.tv = (target.tv || 0) + 1; // sessies van deze gebruiker ongeldig maken saveJson('users.json', users); logEvent(admin, 'admin_reset_password', req, 'Wachtwoord gereset voor: ' + target.email); send(res, 200, { ok: true }); } async function apiDeleteUser(req, res, admin) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const target = findUserById(String(body.userId || '')); if (!target) return send(res, 404, { error: 'Gebruiker niet gevonden.' }); if (target.id === admin.id) return send(res, 400, { error: 'Je kunt jezelf niet verwijderen.' }); if (target.role === 'admin' && countAdmins() === 1) { return send(res, 400, { error: 'Er moet altijd minimaal één beheerder blijven.' }); } users = users.filter(function (u) { return u.id !== target.id; }); delete content[target.id]; saveJson('users.json', users); saveJson('content.json', content); logEvent(admin, 'user_deleted', req, 'Gebruiker verwijderd: ' + target.email); send(res, 200, { ok: true }); } /* ================= API: OVERIG ================= */ function apiLogs(req, res, user) { const isAdmin = user.role === 'admin'; const relevant = isAdmin ? logs : logs.filter(function (l) { return l.userId === user.id; }); const sorted = relevant.slice().reverse().slice(0, 200); send(res, 200, { scope: isAdmin ? 'all' : 'own', logs: sorted }); } function apiGetContent(req, res, user) { const c = ensureContent(user.id); send(res, 200, { countdowns: c.countdowns, todos: c.todos }); } async function apiPutContent(req, res, user) { const body = await readBody(req); if (!body) return send(res, 400, { error: 'Ongeldig verzoek.' }); const countdowns = sanitizeCountdowns(body.countdowns); const todos = sanitizeTodos(body.todos); if (countdowns === null || todos === null) { return send(res, 400, { error: 'Ongeldige gegevens.' }); } content[user.id] = { countdowns: countdowns, todos: todos }; saveJson('content.json', content); send(res, 200, { ok: true }); } /* ================= ROUTING ================= */ async function handleApi(req, res, url) { const route = url.pathname; const user = currentUser(req); if (req.method === 'POST' && route === '/api/register') return await apiRegister(req, res); if (req.method === 'POST' && route === '/api/login') return await apiLogin(req, res); if (req.method === 'POST' && route === '/api/logout') return apiLogout(req, res, user); if (req.method === 'GET' && route === '/api/me') { if (!user) return send(res, 401, { error: 'Niet ingelogd.' }); return send(res, 200, { user: publicUser(user) }); } // Alle verdere eindpunten vereisen een geldige sessie. if (!user) return send(res, 401, { error: 'Niet ingelogd.' }); if (req.method === 'GET' && route === '/api/logs') return apiLogs(req, res, user); if (req.method === 'GET' && route === '/api/content') return apiGetContent(req, res, user); if (req.method === 'PUT' && route === '/api/content') return await apiPutContent(req, res, user); if (req.method === 'POST' && route === '/api/password') return await apiChangePassword(req, res, user); // Gebruikersbeheer: alleen beheerders. if (req.method === 'GET' && route === '/api/users' && requireAdmin(res, user)) return apiListUsers(req, res, user); if (req.method === 'POST' && route === '/api/users/role' && requireAdmin(res, user)) return await apiSetRole(req, res, user); if (req.method === 'POST' && route === '/api/users/reset' && requireAdmin(res, user)) return await apiAdminResetPassword(req, res, user); if (req.method === 'POST' && route === '/api/users/delete' && requireAdmin(res, user)) return await apiDeleteUser(req, res, user); return send(res, 404, { error: 'Onbekend eindpunt.' }); } /* ================= STatische BESTANDEN ================= */ const MIME = { '.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8', '.js': 'text/javascript; charset=utf-8', '.json': 'application/json; charset=utf-8', '.png': 'image/png', '.svg': 'image/svg+xml', '.ico': 'image/x-icon' }; function serveStatic(req, res, url) { if (req.method !== 'GET' && req.method !== 'HEAD') { return send(res, 405, { error: 'Methode niet toegestaan.' }); } let pathname; try { pathname = decodeURIComponent(url.pathname); } catch (e) { return send(res, 400, { error: 'Ongeldig pad.' }); } if (pathname === '/') pathname = '/index.html'; const filePath = path.normalize(path.join(PUBLIC_DIR, pathname)); } catch (e) { /* verbinding al gesloten */ } }); return; } serveStatic(req, res, url); }); server.listen(PORT, function () { console.log(''); console.log(' Portaal & Countdown Beheer Pro V3 draait op:'); console.log(' http://localhost:' + PORT); console.log(''); console.log(' Nieuw in V3: wachtwoord wijzigen, gebruikersbeheer'); console.log(' en wachtwoord-reset door de beheerder.'); console.log('');
